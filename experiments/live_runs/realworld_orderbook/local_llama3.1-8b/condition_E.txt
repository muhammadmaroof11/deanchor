// order_factory.ts
import { OrderBook } from './entities';

export class OrderFactory {
  private orderBook: OrderBook;

  constructor(orderBook: OrderBook) {
    this.orderBook = orderBook;
  }

  createOrder(options: CreateOrderOptions) {
    const { type, side, size, price } = options;
    let order: LimitOrder | StopLimitOrder | StopMarketOrder;

    if (type === 'LIMIT') {
      order = new LimitOrder(size, price);
    } else if (type === 'STOP_LIMIT') {
      order = new StopLimitOrder(size, price, price);
    } else if (type === 'STOP_MARKET') {
      order = new StopMarketOrder(size);
    }

    this.orderBook.orders.push(order);
    return order;
  }
}
