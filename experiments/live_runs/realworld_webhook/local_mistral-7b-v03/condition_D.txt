```typescript
import express from 'express';
import helmet from 'helmet';
import bodyParser from 'body-parser';
import cors from 'cors';
import rateLimit from 'express-rate-limit';
import openapi from 'openapi-generator-client';
import dotenv from 'dotenv';

// Load environment variables from .env file
dotenv.config();

// Initialize the Express app
const app = express();

// Configure rate limiter
const rateLimiter = rateLimit({
  windowMs: 60 * 1000, // 1 minute
  max: 60, // limit each IP to 60 requests per windowMs
});

// App Express configuration
app.use(cors());
app.use(helmet());
app.use(bodyParser.json());

// Set options for Express
app.disable('x-powered-by');

// Define the GitHub webhook route
app.post('/', (req, res) => {
  webhookHandler(req, res);
});

// Define the route to get a list of all the configured routes
app.get('/routes', (req, res) => {
  listRouteHandler(req, res);
});

// Apply rate limiting to the OpenAPI route
app.use('/openapi.json', rateLimiter);
app.get('/openapi.json', (req, res) => {
  openapiHandler(req, res);
});

// Define health check routes
app.post('/health', (req, res) => {
  res.status(200).send('alive');
});
app.post('/dummy1', (req, res) => {
  res.status(200).send('dummy1');
});
app.post('/dummy2', (req, res) => {
  res.status(200).send('dummy2');
});

// Start the server
const port = process.env.PORT || 3000;
app.listen(port, () => {
  console.log(`GitHub webhook dispatcher listening on port ${port}!`);
  console.log(`Debug mode: ${process.env.DEBUG}`);
});

}
```
