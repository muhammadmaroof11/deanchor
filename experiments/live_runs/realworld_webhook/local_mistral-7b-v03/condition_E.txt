3. Implement the main `WebhookDispatcher` file:

```javascript
const express = require('express');
const helmet = require('helmet');
const bodyParser = require('body-parser');
const rateLimit = require('rate-limit');
const swaggerUi = require('swagger-ui-express');
const openapiSpec = require('./openapi.json');

const config = require('./config');
const lib = require('./lib');

const app = express();

app.use(helmet());
app.use(bodyParser.json());
app.use(bodyParser.urlencoded({ extended: true }));

// Rate Limiting
const limiter = rateLimit({
  windowMs: 60 * 1000,
  max: 60,
});
app.use('/openapi.json', limiter, swaggerUi.serve, swaggerUi.setup(openapiSpec));

app.post('/', limiter, lib.webhookHandler);
app.get('/routes', lib.listRouteHandler);
app.post('/health', (req, res) => res.send('alive'));
app.post('/dummy1', (req, res) => res.send('dummy1'));
app.post('/dummy2', (req, res) => res.send('dummy2'));

const server = app.listen(config.port, () => {
  console.log(`GitHub webhook dispatcher listening on port ${config.port}!`);
});

if (config.debug) {
  const debug = require('debug')('webhook-dispatcher');
  debug(server.address);
}
```
