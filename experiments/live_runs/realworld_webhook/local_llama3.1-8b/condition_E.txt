```python
# routes/openapi.py
from fastapi import FastAPI, APIRouter
from pydantic import BaseModel
from fastapi.responses import JSONResponse

class OpenAPI(BaseModel):
    openapi: str
    info: dict
    paths: dict

router = APIRouter()

@router.get("/openapi")
async def get_openapi() -> JSONResponse:
    openapi_spec = OpenAPI(
        openapi="3.0.2",
        info={"title": "GitHub Webhook Handler", "description": "API specification"},
        paths={}
    )
    return JSONResponse(content=openapi_spec.json(), media_type="application/json")
```
