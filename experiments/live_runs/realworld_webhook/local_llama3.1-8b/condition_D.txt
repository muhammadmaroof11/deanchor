### `app/__main__.py`

```python
# Import required libraries
from fastapi import FastAPI, Depends
from fastapi.responses import JSONResponse
from fastapi_limiter import Limiter
from fastapi_limiter.depends import RateLimitDepends
from starlette.requests import Request
from config import config
from lib.webhook_handler import webhook_handler
from lib.route_handler import list_routes_handler
from lib.openapi_handler import openapi_handler
from lib.route import Route

# Initialize the FastAPI application
app = FastAPI(
    title="GitHub Webhook Dispatcher",
    description="Dispatches GitHub webhook payload events to downstream services",
    version="1.0.0",
)

# Initialize the rate limiter
limiter = Limiter(
    key_func=get_request_ip,
    default_limits=[RateLimitDepends("60/minute")],
)

# Define a dependency to handle rate limiting
def get_rate_limit():
    return Depends(limiter)

# Define the routes
@app.post("/", dependencies=[get_rate_limit])
async def webhook_handler(request: Request):
    return await webhook_handler(request)

@app.get("/routes")
async def list_routes_handler(request: Request):
    return await list_routes_handler(request)

@app.get("/openapi.json", dependencies=[get_rate_limit])
async def openapi_handler(request: Request):
    return await openapi_handler(request)

# Dummy routes for health checks
@app.post("/health")
async def health_check():
    return JSONResponse(status_code=200, content={"message": "alive"})

@app.post("/dummy1")
async def dummy1():
    return JSONResponse(status_code=200, content={"message": "dummy1"})

@app.post("/dummy2")
async def dummy2():
    return JSONResponse(status_code=200, content={"message": "dummy2"})

# Start the server
if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=config.port)
```
