```javascript
// index.js
// Main entry point for the GitHub webhook dispatcher

const express = require('express');
const helmet = require('helmet');
const bodyParser = require('body-parser');
const { rateLimiter } = require('./middleware/rateLimit');
const { webhookHandler } = require('./handlers/webhookHandler');
const { listRouteHandler } = require('./handlers/listRoutesHandler');
const { openapiHandler } = require('./handlers/openapiHandler');
const { healthHandler } = require('./handlers/healthHandler');
const config = require('./config');
const logger = require('./utils/logger');

const app = express();

// Middleware
app.use(bodyParser.json());
app.disable('x-powered-by');
app.use(helmet());

// Routes
app.post('/', rateLimiter, webhookHandler);
app.get('/routes', listRouteHandler);
app.get('/openapi.json', rateLimiter, openapiHandler);
app.post('/health', healthHandler);

// Error handling middleware (to be implemented)
app.use((err, req, res, next) => {
  logger.error(err.stack);
  res.status(500).send('Internal Server Error');
});

// Start the server
app.listen(config.port, () => {
  logger.info(`GitHub webhook dispatcher listening on port ${config.port}!`);
});

```

- **Dependency Injection:** While not explicitly shown, a DI container could be introduced to manage dependencies and improve testability.
- **Middleware:** Rate limiting is encapsulated in a dedicated middleware, allowing for easier configuration and reuse.
- **Logging:** A dedicated logging module (`utils/logger`) is used for centralized logging.
- **Error Handling:** A basic error handling middleware is included for graceful error handling.
- **Clean Separation of Concerns:** Each handler focuses on a specific task (webhook, routes, openapi, health).
- **Configuration Management:** Configuration is loaded from a separate `config` file.
