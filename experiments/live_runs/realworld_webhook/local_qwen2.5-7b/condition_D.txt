#### `app.ts`

```typescript
import fastify from 'fastify';
import fastifyRateLimit from 'fastify-rate-limit';
import { config } from './config';
import { webhookHandler, listRouteHandler, openapiHandler } from './lib';
import { logger } from './lib/logger';

const app = fastify({ logger: logger.child({ service: 'app' }) });

// Apply rate limiting to the openapi.json route
const rateLimiter = fastifyRateLimit({
  max: 60,
  windowMS: 60000, // 1 minute
  message: 'Too many requests, please try again later.',
});

app.register(rateLimiter, { prefix: '/openapi.json' });

// Parse the request body as JSON
app.register(require('fastify-bodyparser'));

// Set options for Express
app.register(require('fastify-helmet'));

// Create a route for the GitHub webhook
app.post('/', async (req, res) => {
  try {
    await webhookHandler(req, res);
  } catch (error) {
    logger.error('Error handling webhook', error);
    res.statusCode = 500;
    res.send('Error handling webhook');
  }
});

// Create a route to get a list of all the configured routes
app.get('/routes', async (req, res) => {
  try {
    await listRouteHandler(req, res);
  } catch (error) {
    logger.error('Error listing routes', error);
    res.statusCode = 500;
    res.send('Error listing routes');
  }
});

// Create a route to serve the openapi spec
app.get('/openapi.json', async (req, res) => {
  try {
    await openapiHandler(req, res);
  } catch (error) {
    logger.error('Error handling openapi request', error);
    res.statusCode = 500;
    res.send('Error handling openapi request');
  }
});

// Create a dummy route for health checks
app.post('/health', async (req, res) => {
  res.statusCode = 200;
  res.send('alive');
});

// Start the server
app.listen({ port: config.port }, (err, address) => {
  if (err) {
    logger.error('Failed to start the server', err);
    process.exit(1);
  }
  logger.info(`GitHub webhook dispatcher listening on ${address}`);
});

export default app;
```
