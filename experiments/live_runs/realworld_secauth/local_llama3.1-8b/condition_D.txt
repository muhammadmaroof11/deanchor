**Database Configuration**

```javascript

const { Sequelize, DataTypes } = require('sequelize');
```

**User Service**

Create a `services/userService.js` file to encapsulate user-related business logic:

```javascript
// src/services/userService.js

const User = require('../models/user');
const Role = require('../models/role');
const bcrypt = require('bcryptjs');

async function createUser(username, email, password, roles) {
  const hashedPassword = await bcrypt.hash(password, 8);
  const user = await User.create({
    username,
    email,
    password: hashedPassword,
  });

  if (roles) {
    const rolePromises = roles.map((roleName) => Role.findOne({ where: { name: roleName } }));
    const rolesFound = await Promise.all(rolePromises);
    user.setRoles(rolesFound);
  }

  return user;
}

async function getUserById(id) {
  return await User.findByPk(id);
}

module.exports = { createUser, getUserById };
```
