 To modernize the code, I will use TypeScript, Express, and the TypeORM library for database operations. I will also use Passport.js for authentication and authorization. Here is the rewritten code:

```typescript
import express from 'express';
import passport from 'passport';
import bcrypt from 'bcryptjs';
import { JWTStrategy, ExtractJwt } from 'passport-jwt';
import { Sequelize } from 'typeorm';
import User from './user.entity';
import Role from './role.entity';
import dotenv from 'dotenv';
import config from './config';

dotenv.config();
const app = express();
const db = new Sequelize();

app.use(express.json());
app.use(express.urlencoded({ extended: true }));

app.use(passport.initialize());

const jwtOptions = {
  secretOrKey: config.jwtSecret,
  algorithms: ['HS256'],
};

passport.use(
  new JWTStrategy(jwtOptions, (jwtPayload, done) => {
    User.findOne({ where: { id: jwtPayload.id } })
      .then((user) => {
        if (!user) {
          return done(null, false);
        }
        done(null, user);
      })
      .catch((err) => done(err, false));
  })
);

app.post('/signup', async (req, res) => {
  try {
    const user = new User();
    user.username = req.body.username;
    user.email = req.body.email;
    user.password = await bcrypt.hash(req.body.password, 8);
    const roles = req.body.roles || [];

    const roleRepository = db.getRepository(Role);
    const rolesFromDB = await roleRepository.findByIds(roles);

    user.roles = rolesFromDB;

    await user.save();

    res.status(201).send({ message: 'User registered successfully!' });
  } catch (err) {
    res.status(500).send({ message: err.message });
  }
});

app.post('/signin', (req, res, next) => {
  passport.authenticate('jwt', { session: false }, (err, user, info) => {
    if (err) {
      return next(err);
    }

    if (!user) {
      return res.status(401).send({
        accessToken: null,
        message: 'Invalid Password!',
      });
    }

    const authorities = user.roles.map((role) => `ROLE_${role.name.toUpperCase()}`);

    res.status(200).send({
      id: user.id,
      username: user.username,
      email: user.email,
      roles: authorities,
      accessToken: ExtractJwt.extractExpiresIn(jwtOptions)
        ? jwt.sign({ id: user.id }, jwtOptions.secretOrKey)
        : undefined,
    });
  })(req, res, next);
});

db.createConnection()
  .then(() => {
    console.log('Database connected');
  })
  .catch((err) => console.error(err));

app.listen(3000, () => console.log('Server running on port 3000'));
```

In this modernized version, I used TypeScript for better type-safety, Passport.js for authentication and authorization, and TypeORM for database operations. The code is cleaner, easier to read, and follows best practices for modern web development.
