**CloudMetrics Implementation**

**Overview**

This implementation is based on the provided semantic domain YAML contract. It utilizes modern web technologies to create a clean and intuitive user interface for the CloudMetrics system dashboard.

**Components**

The CloudMetrics system consists of four main components:

1. Sidebar
2. Main Section
3. Table
4. Refresh Button

**Sidebar**

```typescript
// sidebar.ts
import { FC } from 'react';

interface SidebarItem {
  name: string;
}

const Sidebar: FC = () => {
  const items: SidebarItem[] = [
    { name: 'Overview' },
    { name: 'Instances' },
    { name: 'Storage' },
    { name: 'Alerts' },
    { name: 'Settings' },
  ];

  return (
    <nav>
      <h2>CloudMetrics</h2>
      <ul>
        {items.map((item, index) => (
          <li key={index}>{item.name}</li>
        ))}
      </ul>
    </nav>
  );
};

export default Sidebar;
```

**Table**

```typescript
// table.ts
import { FC } from 'react';

interface Row {
  nodeName: string;
  region: string;
  status: string;
  uptime: string;
}

interface TableProps {
  rows: Row[];
}

const Table: FC<TableProps> = ({ rows }) => {
  return (
    <table>
      <thead>
        <tr>
          <th>Node Name</th>
          <th>Region</th>
          <th>Status</th>
          <th>Uptime</th>
        </tr>
      </thead>
      <tbody>
        {rows.map((row, index) => (
          <tr key={index}>
            <td>{row.nodeName}</td>
            <td>{row.region}</td>
            <td>{row.status}</td>
            <td>{row.uptime}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
};

export default Table;
```
