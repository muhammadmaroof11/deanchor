**scripts.js** (in js/scripts.js file)
```javascript
import $ from 'jquery';
import 'aos';

$(document).ready(function() {
    // Initialize AOS
    AOS.init({
        easing: 'ease-in-out',
        duration: 600
    });

    // Initialize Owl Carousel
    $('.owl-carousel').owlCarousel({
        loop: true,
        margin: 20,
        nav: true,
        responsive: {
            0: {
                items: 1
            },
            600: {
                items: 1
            },
            1000: {
                items: 1
            }
        }
    });

    // Initialize Bootstrap
    new bootstrap.Carousel(document.getElementById('carousel'), {
        interval: 5000
    });
});
```
